## AIM   
To write a yacc program to recognize a valid arithmetic expression that uses operator +,- ,* and /.

## ALGORITHM
1.	Start the program.
2.	Write a program in the vi editor and save it with .l extension.
3.	In the lex program, write the translation rules for the operators =,+,-,*,/ and for the identifier.
4.	Write a program in the vi editor and save it with .y extension.
5.	Compile the lex program with lex compiler to produce output file as lex.yy.c. eg $ lex filename.l
6.	Compile the yacc program with yacc compiler to produce output file as y.tab.c. eg $ yacc –d arith_id.y
7.	Compile these with the C compiler as gcc lex.yy.c y.tab.c
8.	Enter an arithmetic expression as input and the tokens are identified as output.

## PROGRAM
```


#include <stdio.h>
#include <ctype.h>
#include <string.h>

int isIdentifier(char str[])
{
    if(!(isalpha(str[0]) || str[0] == '_'))
        return 0;

    for(int i = 1; str[i] != '\0'; i++)
    {
        if(!(isalnum(str[i]) || str[i] == '_'))
            return 0;
    }

    return 1;
}

int isOperator(char ch)
{
    return (ch == '+' || ch == '-' ||
            ch == '*' || ch == '/' ||
            ch == '=');
}

int main()
{
    char exp[100];
    char token[50];
    int i = 0, j = 0;

    printf("Enter Arithmetic Expression:\n");
    fgets(exp, sizeof(exp), stdin);

    printf("\nTOKENS:\n");

    while(exp[i] != '\0')
    {
        /* Ignore spaces */
        if(exp[i] == ' ' || exp[i] == '\t' || exp[i] == '\n')
        {
            i++;
            continue;
        }

        /* Identifier */
        if(isalpha(exp[i]) || exp[i] == '_')
        {
            j = 0;

            while(isalnum(exp[i]) || exp[i] == '_')
            {
                token[j++] = exp[i++];
            }

            token[j] = '\0';

            if(isIdentifier(token))
            {
                printf("Identifier : %s\n", token);
            }
        }

        /* Number */
        else if(isdigit(exp[i]))
        {
            j = 0;

            while(isdigit(exp[i]))
            {
                token[j++] = exp[i++];
            }

            token[j] = '\0';

            printf("Number     : %s\n", token);
        }

        /* Operator */
        else if(isOperator(exp[i]))
        {
            switch(exp[i])
            {
                case '=':
                    printf("Operator   : EQUAL (=)\n");
                    break;

                case '+':
                    printf("Operator   : PLUS (+)\n");
                    break;

                case '-':
                    printf("Operator   : MINUS (-)\n");
                    break;

                case '*':
                    printf("Operator   : MULTIPLICATION (*)\n");
                    break;

                case '/':
                    printf("Operator   : DIVISION (/)\n");
                    break;
            }

            i++;
        }

        /* Invalid Character */
        else
        {
            printf("Invalid Character : %c\n", exp[i]);
            i++;
        }
    }

    printf("\nValid Arithmetic Expression\n");

    return 0;
}
```
