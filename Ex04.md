## PROGRAM
```


#include <stdio.h>
#include <ctype.h>
#include <string.h>

int isValidVariable(char str[])
{
    int i;

    /* First character must be alphabet or underscore */
    if(!(isalpha(str[0]) || str[0] == '_'))
    {
        return 0;
    }

    /* Remaining characters can be alphabet, digit, underscore */
    for(i = 1; str[i] != '\0'; i++)
    {
        if(!(isalnum(str[i]) || str[i] == '_'))
        {
            return 0;
        }
    }

    return 1;
}

int main()
{
    char variable[100];

    printf("Enter a variable name:\n");
    scanf("%s", variable);

    if(isValidVariable(variable))
    {
        printf("Valid variable: %s\n", variable);
    }
    else
    {
        printf("Invalid variable name\n");
    }

    return 0;
}
```
